-- Codeer.LowCode.Blazor.Extras 承認フロー / メール送信履歴 サンプル用テーブル (PostgreSQL)
-- Extras フレーム配下の 経費申請 / 購買申請 / 承認待ち / 承認状況 / 承認経路マスタ / ユーザー / メール送信履歴 で使用。
-- 承認モジュール群・メール履歴のテーブル定義は デザイナの Tools > 承認フローのセットアップ / メールのセットアップ が生成した DDL。
-- デモサイトは認証を持たないため、操作ユーザーは appsettings の DemoUserId (= app_users.id) で固定している (既定 2 = 佐藤 花子・課長)。

-- ユーザー (Current User Module = AppUser)
CREATE TABLE app_users (
    id SERIAL PRIMARY KEY,
    name TEXT,
    email TEXT
);

-- 申請書: 経費申請 (ApprovalFlowField の FK 列 = approval_id)
CREATE TABLE expense_requests (
    id SERIAL PRIMARY KEY,
    title TEXT,
    amount NUMERIC,
    approval_id TEXT
);

-- 申請書: 購買申請 (assessed_amount = 最終承認者だけが記入できる査定額)
CREATE TABLE purchase_requests (
    id SERIAL PRIMARY KEY,
    title TEXT,
    amount NUMERIC,
    approval_id TEXT,
    assessed_amount NUMERIC
);

-- 承認フロー (生成 DDL)
CREATE TABLE approval_flows (
  id BIGSERIAL PRIMARY KEY,
  status TEXT,
  target_module_name TEXT,
  target_id TEXT,
  applicant TEXT,
  attempt_no INTEGER,
  current_step_no INTEGER,
  version INTEGER
);
CREATE TABLE approval_flow_members (
  id BIGSERIAL PRIMARY KEY,
  flow_id TEXT,
  attempt_no INTEGER,
  step_no INTEGER,
  step_name TEXT,
  step_type TEXT,
  completion_policy TEXT,
  is_comment_required_on_reject BOOLEAN,
  return_scope TEXT,
  approver_user TEXT,
  is_required BOOLEAN,
  is_final_step BOOLEAN,
  status TEXT,
  acted_at TIMESTAMP
);
CREATE TABLE approval_histories (
  id BIGSERIAL PRIMARY KEY,
  flow_id TEXT,
  attempt_no INTEGER,
  action TEXT,
  actor_user TEXT,
  comment TEXT,
  acted_at TIMESTAMP
);
CREATE TABLE approval_routes (
  id BIGSERIAL PRIMARY KEY,
  route_name TEXT NOT NULL
);
CREATE TABLE approval_route_steps (
  id BIGSERIAL PRIMARY KEY,
  approval_route_id TEXT,
  step_no INTEGER,
  step_name TEXT NOT NULL,
  step_type TEXT,
  completion_policy TEXT,
  is_comment_required BOOLEAN,
  return_scope TEXT
);
CREATE TABLE approval_route_step_members (
  id BIGSERIAL PRIMARY KEY,
  approval_route_step_id TEXT,
  approver_user TEXT NOT NULL,
  is_required BOOLEAN
);

-- メール送信履歴 (生成 DDL)
CREATE TABLE mail_histories (
  id BIGSERIAL PRIMARY KEY,
  sent_at TIMESTAMP,
  mail_infra_name TEXT,
  subject TEXT,
  total_count INTEGER,
  success_count INTEGER,
  failure_details TEXT,
  source_module TEXT,
  source_id TEXT
);
CREATE TABLE mail_history_details (
  id BIGSERIAL PRIMARY KEY,
  history_id TEXT,
  to_address TEXT,
  subject TEXT,
  body TEXT,
  is_success BOOLEAN,
  error TEXT
);

-- お知らせメール (MailField / BulkMailField のデモ)
CREATE TABLE mail_notices (
    id SERIAL PRIMARY KEY,
    title TEXT,
    body TEXT,
    to_address TEXT
);
CREATE TABLE mail_notice_recipients (
    id SERIAL PRIMARY KEY,
    notice_id TEXT,
    user_id TEXT
);
