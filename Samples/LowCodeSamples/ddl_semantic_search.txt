-- 意味検索 (SemanticSearchField) のデモ: 問い合わせ (ExtrasSemanticInquiry)
-- PostgreSQL + pgvector。search_text / search_vector はアプリが保存時に書き、search_vector_v はそれをキャストする生成列 (距離の計算に使う)。
-- ここで入れる行には search_text / search_vector が無いので、入れた後に索引を付ける (SemanticSearchField の Reindex()、またはデモ作成時と同じく埋め込みを計算して UPDATE)。

CREATE EXTENSION IF NOT EXISTS vector;

CREATE TABLE IF NOT EXISTS semantic_inquiries (
    id              SERIAL PRIMARY KEY,
    subject         TEXT,
    body            TEXT,
    response        TEXT,
    search_text     TEXT,
    search_vector   TEXT,
    search_vector_v vector(1536) GENERATED ALWAYS AS (search_vector::vector) STORED
);
CREATE INDEX IF NOT EXISTS semantic_inquiries_vector_idx ON semantic_inquiries USING hnsw (search_vector_v vector_cosine_ops);
