-- 認可 (権限) のデモ用の社員名簿 (PostgreSQL)
-- サイドバー「セキュリティ > 認可 (権限)」(DemoAuthorization) で使用。
-- 同じ表を「条件なしの画面 (DemoAuthzEmployeeAll)」と「権限の条件を付けた画面 (DemoAuthzEmployee)」で並べて表示する。
-- ログインする社員は app_user_id で app_users.id (AppUser) と対応する (ログインしない社員は NULL)。上長 (manager_user_id) も app_users.id。
-- 「自分と同じ部署の行だけ」の判定には、ログインユーザーの所属部署の表 (demo_authz_user_departments) を使う。
-- デモの操作ユーザーは 佐藤 花子 (app_users.id = 2・営業部 課長) に固定されている。
-- 何度流しても同じ (表があれば作らない・行が 1 行でもあれば入れない)。
--   LowCodeSamples.Designer.exe sql "<Design フォルダ>" --datasource Main --file ddl_authorization_demo.txt

CREATE TABLE IF NOT EXISTS demo_authz_employees (
  id serial PRIMARY KEY,
  name text NOT NULL,
  department text NOT NULL,
  title text,
  app_user_id text,
  manager_user_id text,
  monthly_salary numeric,
  evaluation text
);

  id serial PRIMARY KEY,
  app_user_id text NOT NULL,

);
