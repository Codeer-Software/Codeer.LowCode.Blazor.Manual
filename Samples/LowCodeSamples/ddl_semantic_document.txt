--        LowCodeSamples.Designer.exe sql "<Design フォルダ>" --datasource Main --file ddl_semantic_document.txt

--      (SemanticSearchField の ReindexMissing。添付の中身を取り出して文章を作り、埋め込みベクトルまで付ける。索引のある行は何もしない)

--   「権限つき AI チャット」(ModuleDataAccess) は画面と同じ権限で読むので GRANT は要らない (文章の列は書き込み専用なので、見つけた文書のリンクまでを答える)。

CREATE EXTENSION IF NOT EXISTS vector;

CREATE TABLE IF NOT EXISTS semantic_documents (
    id              SERIAL PRIMARY KEY,
    title           TEXT,
    category        TEXT,
    file_name       TEXT,
    file_size       BIGINT,
    file_guid       UUID,
    file_content    BYTEA,
    search_text     TEXT,
    search_vector   TEXT,
    search_vector_v vector(1536) GENERATED ALWAYS AS (search_vector::vector) STORED
);
CREATE INDEX IF NOT EXISTS semantic_documents_vector_idx ON semantic_documents USING hnsw (search_vector_v vector_cosine_ops);
