-- Codeer.LowCode.Blazor.Extras タグ (TagField) サンプル用テーブル (PostgreSQL)
-- サイドバー「業務でよく使う機能 > タグ > 得意先」で使用。
-- tag_customer_tags は デザイナの Tools > タグのセットアップ (tag-setup --target TagCustomer) が生成した DDL。

-- 得意先 (TagField を置いたモジュール)
CREATE TABLE tag_customers (
    id SERIAL PRIMARY KEY,
    name TEXT,
    industry TEXT,
    owner TEXT,
    note TEXT
);

-- タグ付け (1 行 = 得意先に付いたタグ 1 つ。得意先を消せば一緒に消える)
CREATE TABLE tag_customer_tags (
  id BIGSERIAL PRIMARY KEY,
  owner_id BIGINT NOT NULL,
  name VARCHAR(200) NOT NULL
);
CREATE UNIQUE INDEX ux_tag_customer_tags_owner_id_name ON tag_customer_tags (owner_id, name);
CREATE INDEX ix_tag_customer_tags_name ON tag_customer_tags (name);
ALTER TABLE tag_customer_tags ADD CONSTRAINT fk_tag_customer_tags_owner_id FOREIGN KEY (owner_id) REFERENCES tag_customers (id) ON DELETE CASCADE;
